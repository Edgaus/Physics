import numpy as np
import plotly.graph_objects as go

# Matriz N x 2: cada fila es un vector [x, y]
origenes = np.array([[0, 0], [0, 0], [0, 0]])
vectores = np.array([[2, 3], [3, -1], [-1, 2]])
destinos = origenes + vectores

fig = go.Figure()

# 1. Dibujar TODOS los puntos de origen de golpe pasando los arrays de NumPy
fig.add_trace(
    go.Scatter(
        x=origenes[:, 0],  # Todas las coordenadas X de origen
        y=origenes[:, 1],  # Todas las coordenadas Y de origen
        mode="markers",
        marker=dict(
            color="crimson",
            size=10,
            line=dict(width=1, color="white"),
        ),
        name="Orígenes",
    )
)

# 2. Agregar cada flecha iterando sobre las filas de los arrays
for orig, dest in zip(origenes, destinos):
  fig.add_annotation(
      x=dest[0],
      y=dest[1],
      ax=orig[0],
      ay=orig[1],
      xref="x",
      yref="y",
      axref="x",
      ayref="y",
      showarrow=True,
      arrowhead=2,
      arrowsize=1.5,
      arrowwidth=2,
      arrowcolor="#1f77b4",
  )

fig.update_layout(xaxis=dict(range=[-1, 5]), yaxis=dict(range=[-1, 5]))
fig.show()
