import numpy as np
from scipy.interpolate import UnivariateSpline, interp1d
from scipy.optimize import differential_evolution
import pandas as pd
import matplotlib.pyplot as plt
from scipy.interpolate import interp1d as interp1

def complex_fresnel_matrix(wave, n_dispersive, thickness, theta0, polar):
    wave = np.atleast_1d(wave)
    W = len(wave)
    n = np.asarray(n_dispersive, dtype=complex)
    thickness = np.atleast_1d(thickness)

    # Initialize transfer matrices
    M_P = np.tile(np.eye(2, dtype=complex), (W, 1, 1))
    M_S = np.tile(np.eye(2, dtype=complex), (W, 1, 1))

    for m in range(1, n.shape[0] - 1):
        sin_theta_j = (n[0, :] / n[m, :]) * np.sin(theta0)
        cos_theta_j = np.sqrt(1 - sin_theta_j**2 + 0j)

        d_m = thickness[m-1] 
        beta = (2 * np.pi / wave) * d_m * n[m, :] * cos_theta_j

        Mp = np.zeros((W, 2, 2), dtype=complex)
        Mp[:, 0, 0] = np.cos(beta)
        Mp[:, 0, 1] = 1j * cos_theta_j * np.sin(beta) / n[m, :]
        Mp[:, 1, 0] = 1j * n[m, :] * np.sin(beta) / cos_theta_j
        Mp[:, 1, 1] = np.cos(beta)

        Ms = np.zeros((W, 2, 2), dtype=complex)
        Ms[:, 0, 0] = np.cos(beta)
        Ms[:, 0, 1] = 1j * np.sin(beta) / (n[m, :] * cos_theta_j)
        Ms[:, 1, 0] = 1j * n[m, :] * np.sin(beta) * cos_theta_j
        Ms[:, 1, 1] = np.cos(beta)

        M_P = M_P @ Mp
        M_S = M_S @ Ms

    cos_theta_0 = np.cos(theta0)
    sin_theta_L = (n[0, :] / n[-1, :]) * np.sin(theta0)
    cos_theta_L = np.sqrt(1 - sin_theta_L**2 + 0j)

    P_in = np.array([[1, cos_theta_0], [1, -cos_theta_0]], dtype=complex)
    P_out = np.zeros((W, 2, 2), dtype=complex)
    P_out[:, 0, 0], P_out[:, 1, 0] = cos_theta_L, n[-1, :]
    A_P = (1/(2*cos_theta_0)) * (P_in @ M_P @ P_out)
    r_P = A_P[:, 1, 0] / A_P[:, 0, 0]

    S_in = np.array([[cos_theta_0, 1], [cos_theta_0, -1]], dtype=complex)
    S_out = np.zeros((W, 2, 2), dtype=complex)
    S_out[:, 0, 0], S_out[:, 1, 0] = 1, cos_theta_L * n[-1, :]
    A_S = (1/(2*cos_theta_0)) * (S_in @ M_S @ S_out)
    r_S = A_S[:, 1, 0] / A_S[:, 0, 0]

    if polar == 's': R = np.abs(r_S)**2
    elif polar == 'p': R = np.abs(r_P)**2
    else: R = 0.5*(np.abs(r_P)**2 + np.abs(r_S)**2)

    # ---------- Extrema Detection ----------
    # FIX: Scale to nanometers so the Spline math stays stable!
    xs = wave * 1e9 
    ys = R
    
    idx = np.argsort(xs)
    xs, ys = xs[idx], ys[idx]

    spline = UnivariateSpline(xs, ys, k=3, s=0)
    dy_dx = spline.derivative()

    x_dense = np.linspace(xs[0], xs[-1], 2000)
    dy_dense = dy_dx(x_dense)

    sign_change = np.where(np.diff(np.sign(dy_dense)))[0]
    critical_points = x_dense[sign_change]

    d2y_dx2 = spline.derivative(n=2)
    maxima_x = [cp for cp in critical_points if d2y_dx2(cp) < 0]
    minima_x = [cp for cp in critical_points if d2y_dx2(cp) > 0]

    # Convert the found roots back to meters for the optimizer
    return 100*R, np.array(maxima_x) * 1e-9, np.array(minima_x) * 1e-9


wave = np.linspace(200, 1000, 1000)  # Wavelength range from 200 nm to 1000 nm


# 1. Load the data and force conversion to a numpy array
# We use .values to extract the data from the DataFrame
aln_n_data = pd.read_csv(r"Fresnel\Data\AlN Film_n.txt", delimiter="\t", header=None, skiprows=1).values
aln_k_data = pd.read_csv(r"Fresnel\Data\AlN Film_k.txt", delimiter="\t", header=None, skiprows=1).values

si_n_data = pd.read_csv(r"Fresnel\Data\Si_n.txt", delimiter="\t", header=None, skiprows=1).values
si_k_data = pd.read_csv(r"Fresnel\Data\Si_k.txt", delimiter="\t", header=None, skiprows=1).values

# 2. Create the complex refractive index: n - ik
# Slicing [:, 1] gets the second column (the actual n or k values)
aln_complex = aln_n_data[:, 1].astype(float) - 1j * aln_k_data[:, 1].astype(float)
si_complex = si_n_data[:, 1].astype(float) - 1j * si_k_data[:, 1].astype(float)

# 3. Interpolate using the first column [:, 0] as the wavelength reference
f_aln = interp1d(aln_n_data[:, 0], aln_complex, kind='cubic', fill_value="extrapolate")
aln_pc = f_aln(wave)

f_si = interp1d(si_n_data[:, 0], si_complex, kind='cubic', fill_value="extrapolate")
si_pc = f_si(wave)







theta0 = 15 * np.pi / 180  # Normal incidence (converted to radians)





n = np.array([np.ones_like(aln_pc), aln_pc, si_pc])  # Air, AlN, Si
thickness = np.array([100])  # Thickness of the AlN layer




R , ma, mi = complex_fresnel_matrix(wave, n, thickness, theta0, polar='m')
plt.plot(wave, R)
plt.scatter(ma, R[np.searchsorted(wave, ma)], color='red', label='Maxima')
plt.scatter(mi, R[np.searchsorted(wave, mi)], color='blue', label='Minima')
plt.xlabel('Wavelength (nm)')
plt.ylabel('Reflectance')
plt.legend()
plt.show()
